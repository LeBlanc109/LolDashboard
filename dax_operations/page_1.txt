PlayerKills = 
VAR SelectedPlayer = SELECTEDVALUE(PlayerSelector[player_gamertag])
VAR PlayerID = 
    CALCULATE(
        MAX(df1[participantId]),
        df1[player_gamertag] = SelectedPlayer
    )
RETURN
    CALCULATE(
        COUNTROWS(df3),
        df3[type] = "CHAMPION_KILL",
        df3[killerId] = PlayerID
    )

PlayerCS = 
VAR SelectedPlayer = SELECTEDVALUE(PlayerSelector[player_gamertag])
VAR MaxFrame = MAX(df1[frame])
RETURN
    CALCULATE(
        MAX(df1[creep_score]),
        df1[player_gamertag] = SelectedPlayer,
        df1[frame] = MaxFrame
    )

PlayerResult = 
VAR SelectedPlayer = SELECTEDVALUE(PlayerSelector[player_gamertag])
VAR PlayerTeam = 
    CALCULATE(
        MAX('df1'[team]),
        'df1'[player_gamertag] = SelectedPlayer
    )
VAR WinningTeam = 
    CALCULATE(
        MAX('df3'[winningTeam]),
        'df3'[type] = "GAME_END"
    )
RETURN
    IF(PlayerTeam = WinningTeam, "WIN", "LOSS")
